cmake_minimum_required(VERSION 3.21)
project(Auto7zCpp VERSION 6.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
qt_add_executable(Auto7zCpp WIN32 src/main.cpp src/engine.cpp src/engine.h src/ui.cpp src/ui.h)
target_link_libraries(Auto7zCpp PRIVATE Qt6::Widgets)
if(MSVC)
 target_compile_options(Auto7zCpp PRIVATE /utf-8 /W4)
endif()

include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(engine_tests tests/engine_tests.cpp src/engine.cpp src/engine.h)
 target_link_libraries(engine_tests PRIVATE Qt6::Test Qt6::Core)
 target_include_directories(engine_tests PRIVATE src)
 if(MSVC)
  target_compile_options(engine_tests PRIVATE /utf-8)
 endif()
 add_test(NAME engine_tests COMMAND engine_tests)
 qt_add_executable(ui_tests tests/ui_tests.cpp src/ui.cpp src/ui.h src/engine.cpp src/engine.h)
 target_link_libraries(ui_tests PRIVATE Qt6::Test Qt6::Widgets)
 target_include_directories(ui_tests PRIVATE src)
 if(MSVC)
  target_compile_options(ui_tests PRIVATE /utf-8)
 endif()
 add_test(NAME ui_tests COMMAND ui_tests)
endif()
